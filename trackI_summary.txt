## 使用方法

1. 创建采样器:
   static SAMPLER: PerformanceSampler = PerformanceSampler::new("name", 100, 1000);

2. 包装函数:
   SAMPLER.sample("call_site", || { your_code() });

3. 获取统计:
   let summary = SAMPLER.summary();
   println!("P95: {} ns", summary.p95);

4. 导出指标:
   let output = export_metrics(&collector, ExportFormat::Prometheus);
